# Randomly oversample the minority class to match the majority class size
def random_oversample(X, y, random_state=42):
    import numpy as np
    np.random.seed(random_state)
    class_counts = y.value_counts()
    majority_class = class_counts.idxmax()
    minority_class = class_counts.idxmin()
    majority_idx = y[y == majority_class].index
    minority_idx = y[y == minority_class].index
    n_majority = len(majority_idx)
    n_minority = len(minority_idx)
    if n_majority == n_minority:
        return X.copy(), y.copy()
    additional_idx = np.random.choice(minority_idx, size=n_majority - n_minority, replace=True)
    oversampled_idx = minority_idx.union(additional_idx)
    combined_idx = majority_idx.union(oversampled_idx)
    X_over = X.loc[combined_idx].reset_index(drop=True)
    y_over = y.loc[combined_idx].reset_index(drop=True)
    return X_over, y_over

# Load and preprocess dataset: drop non-informative columns, convert target to binary, and one-hot encode
def load_and_preprocess(url):
    import pandas as pd
    df = pd.read_csv(url)
    df.columns = df.columns.str.strip().str.replace(' ', '_')
    drop_cols = ['EmployeeNumber', 'Over18', 'StandardHours', 'EmployeeCount']
    for col in drop_cols:
        if col in df.columns:
            df.drop(columns=col, inplace=True)
    df['Attrition'] = df['Attrition'].map({'Yes': 1, 'No': 0})
    df = pd.get_dummies(df, drop_first=True)
    X = df.drop('Attrition', axis=1)
    y = df['Attrition']
    return X, y

# Plot distribution of the target variable
def plot_attrition_distribution(y):
    import matplotlib.pyplot as plt
    import seaborn as sns
    plt.figure(figsize=(5,3))
    sns.countplot(x=y)
    plt.title('Attrition Distribution')
    plt.xticks([0,1],['No','Yes'])
    plt.xlabel('Attrition')
    plt.ylabel('Count')
    plt.show()

# Plot confusion matrix for logistic regression
def plot_confusion_matrix(cm):
    import matplotlib.pyplot as plt
    import seaborn as sns
    plt.figure(figsize=(4,3))
    sns.heatmap(cm, annot=True, fmt='d', cmap='Blues', xticklabels=['No','Yes'], yticklabels=['No','Yes'])
    plt.title('Confusion Matrix (Logistic Regression)')
    plt.xlabel('Predicted')
    plt.ylabel('Actual')
    plt.show()

# Plot feature importances
def plot_feature_importance(importances, feature_names, top_n=10):
    import matplotlib.pyplot as plt
    import pandas as pd
    s = pd.Series(importances, index=feature_names)
    top_features = s.nlargest(top_n).sort_values()
    plt.figure(figsize=(6,3))
    top_features.plot(kind='barh')
    plt.title(f'Top {top_n} Feature Importances')
    plt.xlabel('Importance')
    plt.ylabel('Feature')
    plt.show()

# Train/test split
X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.2, stratify=y, random_state=42)

# Oversample minority class in training data
X_train_over, y_train_over = random_oversample(X_train, y_train)

# Standardize features
scaler = StandardScaler().fit(X_train_over)
X_train_scaled = scaler.transform(X_train_over)
X_test_scaled = scaler.transform(X_test)

# Define models
models = {
    'Logistic Regression': LogisticRegression(max_iter=1000, n_jobs=-1),
    'Decision Tree': DecisionTreeClassifier(random_state=42),
    'Random Forest': RandomForestClassifier(n_estimators=100, random_state=42)
}

results = {}
conf_matrix = None

for name, model in models.items():
    model.fit(X_train_scaled, y_train_over)
    preds = model.predict(X_test_scaled)
    proba = model.predict_proba(X_test_scaled)[:,1] if hasattr(model, 'predict_proba') else None
    results[name] = {
        'Accuracy': accuracy_score(y_test, preds),
        'Precision': precision_score(y_test, preds),
        'Recall': recall_score(y_test, preds),
        'F1-score': f1_score(y_test, preds),
        'ROC-AUC': roc_auc_score(y_test, proba) if proba is not None else np.nan
    }
    if name == 'Logistic Regression':
        conf_matrix = confusion_matrix(y_test, preds)

# Display metrics
import pandas as pd
metrics_df = pd.DataFrame(results).T
metrics_df
